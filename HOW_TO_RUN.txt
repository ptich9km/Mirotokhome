ШАГ 1: СКАЧАЙТЕ GODOT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Откройте: https://godotengine.org/download
2. Скачайте "Godot Engine 4.7" (Standard version)
3. Распакуйте в любую папку (например, C:\Work\Godot)
4. Запустите Godot

ШАГ 2: ОТКРОЙТЕ ПРОЕКТ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. В Godot нажмите "Import"
2. Перейдите в папку проекта: C:\Work\Allodshome
   (на другой машине — свой путь к этой же папке)
3. Нажмите "Import & Edit"
4. Подождите: первый импорт ассетов занимает несколько минут

ШАГ 3: ЗАПУСТИТЕ ИГРУ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Стартовая сцена — scenes/character_select.tscn
   (выбор героя; main.tscn запускается после него)
2. Нажмите F5
3. Создайте героя и нажмите «Начать игру»

УПРАВЛЕНИЕ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
• ЛКМ по земле          → Движение
• ЛКМ по врагу/объекту  → Атака/взаимодействие
• ПКМ                   → Отмена прицеливания
• 1..9                  → Быстрые слоты (1..3 переназначаются)
• Ctrl+1..9             → Назначить заклинание/предмет на слот
• Пробел                → Тактическая пауза
• Esc                   → Меню сохранений/настроек
• I / B / W / M         → Склад / книга магии / журнал / карта мира

ЧТО ВИДЕТЬ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
✅ Сгенерированный сектор: 7 биомов, города, дороги
✅ Герой и туман войны
✅ Серые (10 видов × 5 тиров), наёмники, квесты глав городов
✅ HP/Mana бары, панель магии, миникарта
✅ Сохранения: 6 слотов + автосейв

ЕСЛИ НЕ РАБОТАЕТ:
